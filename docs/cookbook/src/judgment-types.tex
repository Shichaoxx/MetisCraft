\newcommand{\SupMat}[9]{%
  \pgfmathsetmacro{\tw}{#5*\U}\pgfmathsetmacro{\th}{#4*\U}%
  \pgfmathsetmacro{\tx}{#2-\tw/2}\pgfmathsetmacro{\ty}{#3-\th/2}%
  \foreach \v [count=\ri] in {#7}{%
    \foreach \c in {1,...,#5}{%
      \ifnum\c>\v\relax\else
        \pgfmathtruncatemacro{\lv}{mod(int(abs(sin(\ri*127.1+\c*311.7+#5*41.3+#4*7.9))*1000),3)}%
        \pgfmathtruncatemacro{\tone}{\lv==0?30:(\lv==1?55:80)}%
        \path[fill=#6!\tone,rounded corners=.6pt]
          ({\tx+(\c-1)*\U+.022},{\ty+\th-\ri*\U+.022}) rectangle ++({\U-.044},{\U-.044});
      \fi}}%
  \path[draw=Ink!45,line width=.4pt,rounded corners=.8pt] (\tx,\ty) rectangle ++(\tw,\th);
  \coordinate (#1-w) at (\tx,#3);\coordinate (#1-e) at ({\tx+\tw},#3);
  \coordinate (#1-n) at (#2,{\ty+\th});\coordinate (#1-s) at (#2,\ty);
  \node[sym] at (#2,{\ty-.27}) {$#8$};
  \node[shp] at (#2,{\ty-.6}) {$#9$};}
